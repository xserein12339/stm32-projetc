```bash
# OpenOCD 自带 ST-Link 规则
```

### 2.2 获取drivers

#### 2.2.1 HAL库获取

```bash
git clone https://github.com/STMicroelectronics/STM32CubeF1.git
```

​	将项目中文件夹`STM32F1xx_HAL`复制到自己项目中的三方库中。这个库的作用是向自建项目提供库支持，可以选择标准库，hal库，也可以直接操作寄存器。

#### 2.2.2 获取hal库配置文件

​	直接在拉取的仓库中找到项目文件夹，在项目文件夹中将`stm32f1xx_hal_conf_template.h`文件作为配置文件模板进行配置，按照格式进行`stm32f1xx_hal_conf.h`配置文件编写，它是对 HAL 库进行“本地化适配”的唯一合法入口，也是 BSP可移植性的基石。 更换芯片型号或电路板时，只需替换此文件，上层应用代码几乎无需改动。这个文件的作用有：

*  **模块化裁剪**

  - 开启 `HAL_UART_MODULE_ENABLED` → 编译器才会包含 `stm32f1xx_hal_uart.c`

  - 注释掉 `HAL_CAN_MODULE_ENABLED` → CAN 驱动完全不进入最终固件
  - **核心价值：避免将未使用的外设驱动链接进固件，显著节省 Flash 和 RAM。对于 F103C8T6 这种仅 64KB Flash 的芯片尤为关键。**

* **硬件参数注入：**HAL 库是通用的，不知道你的板子上晶振是多少、供电电压多高。该文件将这些板级特定信息以宏的形式注入到通用驱动中：

  * `HSE_VALUE=8000000U` → `SystemCoreClockUpdate()` 据此计算 PLL 倍频系数
  * `VDD_VALUE=3300U` → PWR 模块据此判断低功耗模式下的稳压器配置
  * `LSE_STARTUP_TIMEOUT=5000U` → RTC 初始化时等待 LSE 稳定的最大轮询次数
  * **核心价值：实现“同一套 HAL 源码 + 不同配置文件 = 适配不同硬件平台”，无需修改 ST 官方驱动代码。**

*  **行为策略定制：**除了硬件参数，它还控制 HAL 库的运行时行为模式：

  - `USE_RTOS=0/1`：决定 `HAL_Delay()` 是用 SysTick 轮询还是调用 RTOS 延时
  - `USE_HAL_xxx_REGISTER_CALLBACKS=0/1`：决定外设回调是用弱函数覆盖（省 Flash）还是运行时注册（灵活但占空间）
  - `PREFETCH_ENABLE=1/0`：决定是否开启 Flash 预取缓冲
  - `USE_SPI_CRC=0/1`：决定 SPI 收发是否自动校验 CRC
  - **核心价值：让开发者在不侵入 HAL 源码的前提下，根据裸机/RTOS、性能/空间等需求调整库的行为。**

* **调试与安全机制**
  * `USE_FULL_ASSERT`：开启后，HAL 内部所有 `assert_param()` 会检查参数合法性，非法时触发 `assert_failed()` 回调
  * `TICK_INT_PRIORITY`：设定 SysTick 中断优先级，防止与其他外设中断冲突
  * **核心价值：在开发阶段快速定位参数误用；在发布阶段一键关闭以获得零开销。**



#### 2.2.3 编写CMakeLists.txt文件

​	这个`CMakeLists.txt`文件的编写目的是将drivers编译为一个自己项目依赖的第三方库进行使用。

##### (1) 设置目录变量

* **设置外设驱动源码目录：**

```CMake
set(HAL_DRIVER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/STM32F1xx_HAL/STM32F1xx_HAL_Driver)
```

* **设置内核寄存器定义与内联函数目录：**

```CMake
set(CMSIS_CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/STM32F1xx_HAL/CMSIS/Core)
```

* **设置芯片级寄存器定义、启动文件、系统时钟初始化：**

```Cmake
set(CMSIS_DEVICE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/STM32F1xx_HAL/CMSIS/Device/ST/STM32F1xx)
```



##### (2) 添加接口库

  将`CMSIS Core`、`CMSIS Device`作为纯头文件接口库，他们的作用是提供内核寄存器定义与内联函数、芯片级寄存器定义、系统时钟初始化等接口。

```cmake
# CMSIS Core (纯头文件接口库)
add_library(cmsis_core INTERFACE)
target_include_directories(cmsis_core INTERFACE
    ${CMSIS_CORE_DIR}/Include            # core_cm3.h, cmsis_gcc.h 等
)

# CMSIS Device (芯片头文件、系统时钟初始化、型号宏)
add_library(stm32f1_cmsis_device INTERFACE)
target_include_directories(stm32f1_cmsis_device INTERFACE
    ${CMSIS_DEVICE_DIR}/Include
)
target_link_libraries(stm32f1_cmsis_device INTERFACE cmsis_core)


```



##### (3) 添加外设驱动源码库

```cmake
add_library(stm32f1_hal STATIC
    # --- 基础核心模块 ---
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_cortex.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_rcc.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_gpio.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_dma.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_flash.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_pwr.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_exti.c

    # --- 已启用的外设模块 ---
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_uart.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_spi.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_i2c.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_adc.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_tim.c
    ${HAL_DRIVER_DIR}/Src/stm32f1xx_hal_rtc.c
)

# 添加芯片依赖宏
target_link_libraries(stm32f1_hal PUBLIC
    stm32f103c8tx_target    
)

# 添加头文件
target_include_directories(stm32f1_hal PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}          # stm32f1xx_hal_conf.h 所在目录
    ${HAL_DRIVER_DIR}/Inc                # HAL 驱动头文件
)

# 链接CMSIS库
target_link_libraries(stm32f1_hal PUBLIC
    stm32f1_cmsis_device                 # 自动传递 CMSIS 路径 + STM32F103xB 宏
)

# 定义HAL库开关宏
target_compile_definitions(stm32f1_hal PRIVATE
    USE_HAL_DRIVER                       # HAL 驱动总开关定义
)

# 编译设置
target_compile_options(stm32f1_hal PRIVATE
    -Wall
    -Wno-unused-parameter
)
```

#### 2.3.2 内容

```cmake

# 芯片目标接口库 
add_library(stm32f103c8tx_target INTERFACE)

# CPU架构与浮点选项
target_compile_options(stm32f103c8tx_target INTERFACE
    -mcpu=cortex-m3
    -mthumb
)

# 芯片型号编译宏定义
target_compile_definitions(stm32f103c8tx_target INTERFACE
    STM32F103xB
)

# 链接脚本与链接器选项
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/linker/STM32F103C8Tx_FLASH.ld"
    CACHE FILEPATH "Linker script for STM32F103C8Tx")

target_link_options(stm32f103c8tx_target INTERFACE
    -T${LINKER_SCRIPT}
    -Wl,-Map=${PROJECT_NAME}.map,--cref
    -Wl,--gc-sections
    -Wl,--print-memory-usage
)

# 芯片运行时库 (STATIC，承接启动文件与系统初始化)
add_library(chip_runtime STATIC
    startup/startup_stm32f103xb.s
    system/system_stm32f1xx.c
)

# 获取芯片宏和架构标志
target_link_libraries(chip_runtime PUBLIC stm32f103c8tx_target)

target_include_directories(chip_runtime PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/system             
    ${HAL_DRIVER_DIR}/Inc                           
    ${CMSIS_DEVICE_DIR}/Include                     
    ${CMSIS_CORE_DIR}/Include                   
)

target_link_libraries(chip_runtime PRIVATE stm32f1_hal)
```

```cmake
    COMMAND openocd

        -f target/stm32f1x.cfg
        -c "program $<TARGET_FILE:${PROJECT_NAME}.elf> verify reset exit"
    DEPENDS ${PROJECT_NAME}.elf
```

#### 3.4.2 添加`freertos_kernel`库

```cmake
    ${FREERTOS_DIR}/tasks.c

    ${FREERTOS_DIR}/include
    ${FREERTOS_DIR}/portable/GCC/ARM_CM3
    ${FREERTOS_DIR}/Config               # FreeRTOSConfig.h
)
```

```c
#include "FreeRTOS.h"
#include "task.h"

/*
 * vApplicationGetIdleTaskMemory
 */
#if (configSUPPORT_STATIC_ALLOCATION == 1)
static StaticTask_t xIdleTaskTCB;
static StackType_t  uxIdleTaskStack[configMINIMAL_STACK_SIZE];

void vApplicationGetIdleTaskMemory(StaticTask_t **ppxIdleTaskTCBBuffer,
                                    StackType_t **ppxIdleTaskStackBuffer,
                                    uint16_t *pulIdleTaskStackSize)
{
    *ppxIdleTaskTCBBuffer   = &xIdleTaskTCB;
    *ppxIdleTaskStackBuffer = uxIdleTaskStack;
    *pulIdleTaskStackSize   = configMINIMAL_STACK_SIZE;
}
#endif

/*
 * vApplicationStackOverflowHook
 * 使用内联汇编替代 __BKPT，避免依赖 CMSIS 头文件
 */
#if (configCHECK_FOR_STACK_OVERFLOW > 0)
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName)
{
    (void)xTask;
    (void)pcTaskName;

    /* 栈溢出是致命错误，触发硬件断点等待调试器捕获 */
    __asm volatile("bkpt #0");

    while (1) {}
}
#endif
```
